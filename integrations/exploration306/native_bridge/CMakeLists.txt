cmake_minimum_required(VERSION 3.8)
project(zeng306_frontier_bridge)
find_package(ament_cmake REQUIRED)
find_package(frontier_exploration_ros2 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(candidate_bridge candidate_bridge.cpp)
target_compile_features(candidate_bridge PUBLIC cxx_std_17)
target_compile_options(candidate_bridge PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(candidate_bridge frontier_exploration_ros2::frontier_exploration_ros2_core)
ament_target_dependencies(candidate_bridge rclcpp nav_msgs std_srvs tf2_ros)
install(TARGETS candidate_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
